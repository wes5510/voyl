### **1. 첫 실행 감지 및 모달 표시**

#### **첫 실행 감지 Hook** (`src/renderer/pages/setup/useFirstRun.ts`)

```typescript
import { useEffect, useState } from 'react'
import { checkConfigExists } from '@/renderer/repos/storage'

export function useFirstRun() {
  const [isFirstRun, setIsFirstRun] = useState<boolean | null>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    const checkFirstRun = async () => {
      try {
        const hasConfig = await checkConfigExists()
        setIsFirstRun(!hasConfig)
      } catch (error) {
        console.error('Failed to check first run status:', error)
        // 에러 시 첫 실행으로 간주 (안전한 기본값)
        setIsFirstRun(true)
      } finally {
        setIsLoading(false)
      }
    }

    checkFirstRun()
  }, [])

  return { isFirstRun, isLoading }
}
```

#### **설정 모달 래퍼** (`src/renderer/pages/setup/SetupModal.tsx`)

```tsx
import { useFirstRun } from './useFirstRun'
```

### **2. 메인 저장 위치 선택 모달**

#### **모달 컴포넌트** (`src/renderer/pages/setup/StorageLocationModal.tsx`)

```tsx
import { useState, useEffect } from 'react'
import { RecommendedPath } from '@/renderer/repos/storage'
import RecommendedPaths from './RecommendedPaths'
import CustomPathInput from './CustomPathInput'
import InitializationProgress from './InitializationProgress'

interface StorageLocationModalProps {
  onComplete?: () => void
}

export default function StorageLocationModal({ onComplete }: StorageLocationModalProps) {
  const [selectedPath, setSelectedPath] = useState<string>('')
  const [isCustomPath, setIsCustomPath] = useState(false)
  const [isInitializing, setIsInitializing] = useState(false)
  const [error, setError] = useState<string>('')

  const handlePathSelect = (path: RecommendedPath) => {
    if (path.path === '') {
      // 커스텀 경로 선택
      setIsCustomPath(true)
      setSelectedPath('')
    } else {
      setIsCustomPath(false)
      setSelectedPath(path.path)
    }
    setError('')
  }

  const handleCustomPathChange = (path: string) => {
    setSelectedPath(path)
    setError('')
  }

  const handleInitialize = async () => {
    if (!selectedPath.trim()) {
      setError('Please select a location for your workspace')
      return
    }

    setIsInitializing(true)
    setError('')

    try {
      await initializeWorkspace(selectedPath)
      onComplete?.()
    } catch (error) {
      setError(error.message || 'Failed to initialize workspace')
    } finally {
      setIsInitializing(false)
    }
  }

  return (
    <div className="bg-white rounded-lg shadow-xl w-[600px] max-h-[80vh] overflow-hidden">
      {/* 헤더 */}
      <div className="px-6 py-4 border-b border-gray-200">
        <h2 className="text-xl font-semibold text-gray-900">Welcome to Voyl</h2>
        <p className="text-sm text-gray-600 mt-1">Choose where to store your workspace files</p>
      </div>

      {/* 컨텐츠 */}
      <div className="px-6 py-4 max-h-[400px] overflow-y-auto">
        {isInitializing ? (
          <InitializationProgress />
        ) : (
          <>
            {!isCustomPath ? (
              <RecommendedPaths onSelect={handlePathSelect} />
            ) : (
              <CustomPathInput
                value={selectedPath}
                onChange={handleCustomPathChange}
                onBack={() => setIsCustomPath(false)}
              />
            )}

            {error && (
              <div className="mt-4 p-3 bg-red-50 border border-red-200 rounded-md">
                <p className="text-sm text-red-800">{error}</p>
              </div>
            )}
          </>
        )}
      </div>

      {/* 푸터 */}
      {!isInitializing && (
        <div className="px-6 py-4 border-t border-gray-200 bg-gray-50">
          <div className="flex justify-between items-center">
            <p className="text-xs text-gray-500">{selectedPath && `Selected: ${selectedPath}`}</p>
            <button
              onClick={handleInitialize}
              disabled={!selectedPath.trim()}
              className="px-4 py-2 bg-blue-600 text-white rounded-md hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed"
            >
              Create Workspace
            </button>
          </div>
        </div>
      )}
    </div>
  )
}
```

### **3. 추천 경로 리스트 컴포넌트**

#### **추천 경로 목록** (`src/renderer/pages/setup/RecommendedPaths.tsx`)

```tsx
import { useState, useEffect } from 'react'
import { RecommendedPath, getRecommendedPaths } from '@/renderer/repos/storage'
import { FolderIcon, CloudIcon, FolderOpenIcon } from 'lucide-react'

interface RecommendedPathsProps {
  onSelect: (path: RecommendedPath) => void
}

export default function RecommendedPaths({ onSelect }: RecommendedPathsProps) {
  const [paths, setPaths] = useState<RecommendedPath[]>([])
  const [isLoading, setIsLoading] = useState(true)
  const [selectedIndex, setSelectedIndex] = useState<number>(0)

  useEffect(() => {
    const loadPaths = async () => {
      try {
        const recommendedPaths = await getRecommendedPaths()
        setPaths(recommendedPaths)
      } catch (error) {
        console.error('Failed to load recommended paths:', error)
        // 기본 경로라도 제공
        setPaths([
          {
            name: 'Documents',
            path: '~/Documents/Voyl',
            description: 'Default documents folder',
            isCloudService: false,
            icon: 'folder',
          },
        ])
      } finally {
        setIsLoading(false)
      }
    }

    loadPaths()
  }, [])

  const getIcon = (iconName?: string, isCloudService?: boolean) => {
    if (isCloudService) return <CloudIcon className="w-5 h-5" />
    if (iconName === 'folder-open') return <FolderOpenIcon className="w-5 h-5" />
    return <FolderIcon className="w-5 h-5" />
  }

  const handleSelect = (path: RecommendedPath, index: number) => {
    setSelectedIndex(index)
    onSelect(path)
  }

  if (isLoading) {
    return (
      <div className="space-y-3">
        {[...Array(3)].map((_, i) => (
          <div key={i} className="animate-pulse">
            <div className="h-16 bg-gray-200 rounded-lg"></div>
          </div>
        ))}
      </div>
    )
  }

  return (
    <div className="space-y-2">
      <h3 className="text-sm font-medium text-gray-900 mb-3">Recommended Locations</h3>

      {paths.map((path, index) => (
        <button
          key={index}
          onClick={() => handleSelect(path, index)}
          className={`w-full p-4 rounded-lg border text-left transition-colors ${
            selectedIndex === index
              ? 'border-blue-500 bg-blue-50'
              : 'border-gray-200 hover:border-gray-300 hover:bg-gray-50'
          }`}
        >
          <div className="flex items-start space-x-3">
            <div
              className={`flex-shrink-0 p-2 rounded-md ${
                path.isCloudService ? 'bg-blue-100 text-blue-600' : 'bg-gray-100 text-gray-600'
              }`}
            >
              {getIcon(path.icon, path.isCloudService)}
            </div>

            <div className="flex-1 min-w-0">
              <div className="flex items-center space-x-2">
                <h4 className="text-sm font-medium text-gray-900">{path.name}</h4>
                {path.isCloudService && (
                  <span className="text-xs px-2 py-1 bg-blue-100 text-blue-800 rounded-full">
                    Cloud
                  </span>
                )}
              </div>
              <p className="text-sm text-gray-600 mt-1">{path.description}</p>
              {path.path && <p className="text-xs text-gray-500 mt-1 font-mono">{path.path}</p>}
            </div>
          </div>
        </button>
      ))}
    </div>
  )
}
```

### **4. 커스텀 경로 입력 컴포넌트**

#### **커스텀 경로 선택** (`src/renderer/pages/setup/CustomPathInput.tsx`)

```tsx
import { useState, useEffect } from 'react'
import { ArrowLeftIcon, FolderIcon } from 'lucide-react'
import { validatePath, openDirectoryDialog } from '@/renderer/repos/storage'

interface CustomPathInputProps {
  value: string
  onChange: (path: string) => void
  onBack: () => void
}

export default function CustomPathInput({ value, onChange, onBack }: CustomPathInputProps) {
  const [validation, setValidation] = useState<PathValidationResult | null>(null)
  const [isValidating, setIsValidating] = useState(false)

  // 경로 변경 시 실시간 검증
  useEffect(() => {
    if (!value.trim()) {
      setValidation(null)
      return
    }

    const validateAsync = async () => {
      setIsValidating(true)
      try {
        const result = await validatePath(value)
        setValidation(result)
      } catch (error) {
        setValidation({
          isValid: false,
          error: 'Failed to validate path',
          warnings: [],
        })
      } finally {
        setIsValidating(false)
      }
    }

    // 500ms 디바운스
    const timer = setTimeout(validateAsync, 500)
    return () => clearTimeout(timer)
  }, [value])

  const handleBrowse = async () => {
    try {
      const selectedPath = await openDirectoryDialog()
      if (selectedPath) {
        onChange(selectedPath)
      }
    } catch (error) {
      console.error('Failed to open directory dialog:', error)
    }
  }

  return (
    <div className="space-y-4">
      {/* 뒤로가기 */}
      <button
        onClick={onBack}
        className="flex items-center space-x-2 text-sm text-gray-600 hover:text-gray-900"
      >
        <ArrowLeftIcon className="w-4 h-4" />
        <span>Back to recommended locations</span>
      </button>

      <div>
        <h3 className="text-sm font-medium text-gray-900 mb-3">Custom Location</h3>

        {/* 경로 입력 */}
        <div className="space-y-3">
          <div className="flex space-x-2">
            <input
              type="text"
              value={value}
              onChange={(e) => onChange(e.target.value)}
              placeholder="Enter workspace path..."
              className="flex-1 px-3 py-2 border border-gray-300 rounded-md focus:outline-none focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
            />
            <button
              onClick={handleBrowse}
              className="px-3 py-2 border border-gray-300 rounded-md hover:bg-gray-50 flex items-center space-x-1"
            >
              <FolderIcon className="w-4 h-4" />
              <span>Browse</span>
            </button>
          </div>

          {/* 검증 결과 */}
          {isValidating && (
            <div className="flex items-center space-x-2 text-sm text-gray-500">
              <div className="animate-spin w-4 h-4 border-2 border-gray-300 border-t-blue-600 rounded-full"></div>
              <span>Validating path...</span>
            </div>
          )}

          {validation && !isValidating && (
            <div className="space-y-2">
              {/* 에러 표시 */}
              {!validation.isValid && validation.error && (
                <div className="p-3 bg-red-50 border border-red-200 rounded-md">
                  <p className="text-sm text-red-800">{validation.error}</p>
                </div>
              )}

              {/* 경고 표시 */}
              {validation.warnings.length > 0 && (
                <div className="p-3 bg-yellow-50 border border-yellow-200 rounded-md">
                  {validation.warnings.map((warning, index) => (
                    <p key={index} className="text-sm text-yellow-800">
                      {warning}
                    </p>
                  ))}
                </div>
              )}

              {/* 성공 표시 */}
              {validation.isValid && (
                <div className="p-3 bg-green-50 border border-green-200 rounded-md">
                  <p className="text-sm text-green-800">✓ Path is valid</p>
                </div>
              )}
            </div>
          )}
        </div>
      </div>
    </div>
  )
}
```

### **6. IPC 연결 및 API 함수**

#### **저장 위치 관련 API** (`src/renderer/repos/storage.ts`)

```typescript
const { electronAPI } = window

export interface RecommendedPath {
  name: string
  path: string
  description: string
  isCloudService: boolean
  icon?: string
}

export interface PathValidationResult {
  isValid: boolean
  error?: string
  warnings: string[]
}

/**
 * 추천 저장 경로 목록 조회
 */
export async function getRecommendedPaths(): Promise<RecommendedPath[]> {
  try {
    return await electronAPI.storage.getRecommendedPaths()
  } catch (error) {
    console.error('Failed to get recommended paths:', error)
    throw new Error('Failed to load storage options')
  }
}

/**
 * 경로 검증
 */
export async function validatePath(path: string): Promise<PathValidationResult> {
  try {
    return await electronAPI.storage.validatePath(path)
  } catch (error) {
    console.error('Failed to validate path:', error)
    return {
      isValid: false,
      error: 'Failed to validate path',
      warnings: [],
    }
  }
}

/**
 * 워크스페이스 초기화
 */
export async function initializeWorkspace(path: string): Promise<void> {
  try {
    await electronAPI.storage.initializeWorkspace(path)
  } catch (error) {
    console.error('Failed to initialize workspace:', error)
    throw new Error(error.message || 'Failed to initialize workspace')
  }
}

/**
 * 설정 파일 존재 확인
 */
export async function checkConfigExists(): Promise<boolean> {
  try {
    return await electronAPI.storage.hasConfig()
  } catch (error) {
    console.error('Failed to check config:', error)
    return false
  }
}

/**
 * 디렉터리 선택 다이얼로그
 */
export async function openDirectoryDialog(): Promise<string | null> {
  try {
    return await electronAPI.storage.openDirectoryDialog()
  } catch (error) {
    console.error('Failed to open directory dialog:', error)
    return null
  }
}
```
