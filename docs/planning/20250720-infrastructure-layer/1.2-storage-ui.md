### **1. 첫 실행 감지 및 모달 표시**

#### **첫 실행 감지 Hook** (`src/renderer/pages/setup/useFirstRun.ts`)

```typescript
import { useEffect, useState } from 'react'
import { checkIsInitialized } from '@/renderer/repos/app'

export function useFirstRun() {
  const [isFirstRun, setIsFirstRun] = useState<boolean | null>(null)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    const checkFirstRun = async () => {
      try {
        const isInitialized = await checkIsInitialized()
        setIsFirstRun(!isInitialized)
      } catch (error) {
        console.error('Failed to check first run status:', error)
        // 에러 시 첫 실행으로 간주 (안전한 기본값)
        setIsFirstRun(true)
      } finally {
        setIsLoading(false)
      }
    }

    checkFirstRun()
  }, [])

  return { isFirstRun, isLoading }
}
```

#### **설정 모달 래퍼** (`src/renderer/pages/setup/SetupModal.tsx`)

```tsx
import { useFirstRun } from './useFirstRun'
import StorageLocationModal from './StorageLocationModal'

export default function SetupModal() {
  const { isFirstRun, isLoading } = useFirstRun()

  // 로딩 중에는 아무것도 표시하지 않음
  if (isLoading) return null

  // 첫 실행이 아니면 모달 표시하지 않음
  if (!isFirstRun) return null

  return (
    <div className="fixed inset-0 z-50 bg-black/50 flex items-center justify-center">
      <StorageLocationModal />
    </div>
  )
}
```

### **2. 메인 저장 위치 선택 모달**

#### **단순한 워크스페이스 선택 모달** (`src/renderer/pages/setup/StorageLocationModal.tsx`)

```tsx
import { useState } from 'react'
import { selectWorkspacePath, initializeApp, loadApp } from '@/renderer/repos/app'
import InitializationProgress from './InitializationProgress'
import { FolderIcon } from 'lucide-react'

interface StorageLocationModalProps {
  onComplete?: () => void
}

export default function StorageLocationModal({ onComplete }: StorageLocationModalProps) {
  const [selectedPath, setSelectedPath] = useState<string>('')
  const [isInitializing, setIsInitializing] = useState(false)
  const [error, setError] = useState<string>('')

  const handleSelectPath = async () => {
    try {
      const path = await selectWorkspacePath()
      if (path) {
        setSelectedPath(path)
        setError('')
      }
    } catch (error) {
      setError('Failed to open folder selection dialog')
    }
  }

  const handleInitialize = async () => {
    if (!selectedPath.trim()) {
      setError('Please select a location for your workspace')
      return
    }

    setIsInitializing(true)
    setError('')

    try {
      // 1. 초기화 (설정 생성 + 워크스페이스 폴더 생성)
      await initializeApp(selectedPath)
      // 2. 로드 (DB 초기화 등)
      await loadApp()
      onComplete?.()
    } catch (error) {
      setError(error.message || 'Failed to initialize workspace')
    } finally {
      setIsInitializing(false)
    }
  }

  return (
    <div className="bg-white rounded-lg shadow-xl w-[500px] max-h-[80vh] overflow-hidden">
      {/* 헤더 */}
      <div className="px-6 py-4 border-b border-gray-200">
        <h2 className="text-xl font-semibold text-gray-900">Welcome to Voyl</h2>
        <p className="text-sm text-gray-600 mt-1">Choose where to store your workspace files</p>
      </div>

      {/* 컨텐츠 */}
      <div className="px-6 py-4">
        {isInitializing ? (
          <InitializationProgress />
        ) : (
          <div className="space-y-4">
            <div className="text-center">
              <button
                onClick={handleSelectPath}
                className="inline-flex items-center px-6 py-3 border border-gray-300 rounded-lg hover:bg-gray-50 transition-colors"
              >
                <FolderIcon className="w-5 h-5 mr-2 text-gray-500" />
                <span className="text-gray-700">Select Workspace Location</span>
              </button>
              <p className="text-xs text-gray-500 mt-2">
                Will open in Documents folder by default
              </p>
            </div>

            {selectedPath && (
              <div className="p-3 bg-blue-50 border border-blue-200 rounded-md">
                <p className="text-sm text-blue-800">
                  <strong>Selected:</strong> {selectedPath}
                </p>
              </div>
            )}

            {error && (
              <div className="p-3 bg-red-50 border border-red-200 rounded-md">
                <p className="text-sm text-red-800">{error}</p>
              </div>
            )}
          </div>
        )}
      </div>

      {/* 푸터 */}
      {!isInitializing && (
        <div className="px-6 py-4 border-t border-gray-200 bg-gray-50">
          <div className="flex justify-end">
            <button
              onClick={handleInitialize}
              disabled={!selectedPath.trim()}
              className="px-4 py-2 bg-blue-600 text-white rounded-md hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed"
            >
              Create Workspace
            </button>
          </div>
        </div>
      )}
    </div>
  )
}
```

import { CheckIcon } from 'lucide-react'

interface ProgressStep {

    { id: 'complete', label: 'Initialization complete', completed: false },
  ]

  // 실제 초기화 프로세스와 연동
  useEffect(() => {
    // 초기화 진행률을 실시간으로 받아서 업데이트
    // IPC를 통해 백엔드에서 진행률 정보 수신
  }, [])

  return (
    <div className="space-y-6">
      <div className="text-center">
        <h3 className="text-lg font-medium text-gray-900">Setting up your workspace...</h3>
        <p className="text-sm text-gray-600 mt-1">This will only take a moment</p>
      </div>

      {/* 진행률 바 */}
      <div className="space-y-4">
        <div className="w-full bg-gray-200 rounded-full h-2">
          <div
            className="bg-blue-600 h-2 rounded-full transition-all duration-300"
            style={{ width: `${progress}%` }}
          />
        </div>

        <div className="text-center text-sm text-gray-600">{Math.round(progress)}% complete</div>
      </div>

      {/* 단계별 상태 */}
      <div className="space-y-3">
        {steps.map((step, index) => (
          <div key={step.id} className="flex items-center space-x-3">
            <div
              className={`flex-shrink-0 w-6 h-6 rounded-full flex items-center justify-center ${
                step.completed
                  ? 'bg-green-100 text-green-600'
                  : index === currentStep
                    ? 'bg-blue-100 text-blue-600'
                    : 'bg-gray-100 text-gray-400'
              }`}
            >
              {step.completed ? (
                <CheckIcon className="w-4 h-4" />
              ) : (
                <span className="text-xs font-medium">{index + 1}</span>
              )}
            </div>

            <span
              className={`text-sm ${
                step.completed
                  ? 'text-green-800'
                  : index === currentStep
                    ? 'text-gray-900 font-medium'
                    : 'text-gray-600'
              }`}
            >
              {step.label}
            </span>

            {index === currentStep && !step.completed && (
              <div className="flex-1 flex justify-end">
                <div className="animate-spin w-4 h-4 border-2 border-gray-300 border-t-blue-600 rounded-full"></div>
              </div>
            )}
          </div>
        ))}
      </div>
    </div>
  )
}

### **6. IPC 연결 및 API 함수**

#### **앱 관련 API** (`src/renderer/repos/app.ts`)

```typescript
const { electronAPI } = window

export interface RecommendedPath {
  name: string
  path: string
  description: string
  isCloudService: boolean
  icon?: string
}

export interface PathValidationResult {
  isValid: boolean
  error?: string
  warnings: string[]
}

/**
 * 추천 저장 경로 목록 조회
 */
export async function getRecommendedPaths(): Promise<RecommendedPath[]> {
  try {
    return await electronAPI.storage.getRecommendedPaths()
  } catch (error) {
    console.error('Failed to get recommended paths:', error)
    throw new Error('Failed to load storage options')
  }
}

/**
 * 경로 검증
 */
export async function validatePath(path: string): Promise<PathValidationResult> {
  try {
    return await electronAPI.storage.validatePath(path)
  } catch (error) {
    console.error('Failed to validate path:', error)
    return {
      isValid: false,
      error: 'Failed to validate path',
      warnings: [],
    }
  }
}

/**
 * 앱 초기화 (설정 생성 + 워크스페이스 폴더 생성)
 */
export async function initializeApp(path: string): Promise<void> {
  try {
    await electronAPI.app.initializeApp(path)
  } catch (error) {
    console.error('Failed to initialize app:', error)
    throw new Error(error.message || 'Failed to initialize app')
  }
}

/**
 * 앱 로드 (DB 초기화 등)
 */
export async function loadApp(): Promise<void> {
  try {
    await electronAPI.app.loadApp()
  } catch (error) {
    console.error('Failed to load app:', error)
    throw new Error(error.message || 'Failed to load app')
  }
}

/**
 * 앱 초기화 상태 확인
 */
export async function checkIsInitialized(): Promise<boolean> {
  try {
    return await electronAPI.app.isInitialized()
  } catch (error) {
    console.error('Failed to check initialization status:', error)
    return false
  }
}

/**
 * 워크스페이스 경로 선택 (Documents 초기 경로)
 */
export async function selectWorkspacePath(): Promise<string | null> {
  try {
    return await electronAPI.app.selectWorkspacePath()
  } catch (error) {
    console.error('Failed to open directory dialog:', error)
    return null
  }
}
```
