### **1. 메인 앱 시작 프로세스 통합**

#### **앱 진입점 수정** (`src/main/index.ts`)

```typescript
import { app, shell, BrowserWindow, ipcMain } from 'electron'
import { join } from 'path'
import { electronApp, optimizer, is } from '@electron-toolkit/utils'
import { installExtension, REACT_DEVELOPER_TOOLS } from 'electron-devtools-installer'
import './db/connect.js'
import './ipc/index.js'
import { initializeApp } from './common/paths/workspace.js'

// 앱 준비 상태 플래그
let isAppReady = false
let mainWindow: BrowserWindow | null = null

async function createWindow(): Promise<void> {
  // Create the browser window.
  mainWindow = new BrowserWindow({
    width: 900,
    height: 670,
    show: false, // 초기화 완료 후 표시
    autoHideMenuBar: true,
    webPreferences: {
      preload: join(__dirname, '../preload/index.mjs'),
      sandbox: false,
    },
  })

  mainWindow.on('ready-to-show', () => {
    if (isAppReady) {
      mainWindow?.show()
    }
  })

  mainWindow.webContents.setWindowOpenHandler((details) => {
    shell.openExternal(details.url)
    return { action: 'deny' }
  })

  // HMR for renderer base on electron-vite cli.
  if (is.dev && process.env['ELECTRON_RENDERER_URL']) {
    mainWindow.loadURL(process.env['ELECTRON_RENDERER_URL'])
  } else {
    mainWindow.loadFile(join(__dirname, '../renderer/index.html'))
  }
}

/**
 * 앱 완전 초기화 프로세스
 */
async function initializeAppWithUI(): Promise<void> {
  try {
    // 1. 백엔드 초기화
    await initializeApp()

    // 2. UI 준비 완료
    isAppReady = true

    // 3. 메인 윈도우 표시
    if (mainWindow) {
      mainWindow.show()
    }

    console.log('App initialization completed successfully')
  } catch (error) {
    console.error('App initialization failed:', error)

    if (error.message === 'User workspace selection required') {
      // 사용자 워크스페이스 선택 필요 - UI에서 처리
      isAppReady = true
      if (mainWindow) {
        mainWindow.show()
      }
    } else {
      // 치명적 오류 - 앱 종료
      await showFatalErrorDialog(error.message)
      app.quit()
    }
  }
}

/**
 * 치명적 오류 다이얼로그 표시
 */
async function showFatalErrorDialog(message: string): Promise<void> {
  const { dialog } = await import('electron')
  await dialog.showErrorBox(
    'Application Initialization Failed',
    `Voyl cannot start due to the following error:\n\n${message}\n\nPlease contact support if this problem persists.`,
  )
}

// This method will be called when Electron has finished
// initialization and is ready to create browser windows.
app.whenReady().then(async () => {
  // Set app user model id for windows
  electronApp.setAppUserModelId('com.electron')

  app.on('browser-window-created', (_, window) => {
    optimizer.watchWindowShortcuts(window)
  })

  // Create window first
  await createWindow()

  // Then initialize app
  await initializeAppWithUI()

  app.on('activate', function () {
    if (BrowserWindow.getAllWindows().length === 0) {
      createWindow()
    }
  })

  if (is.dev) {
    installExtension(REACT_DEVELOPER_TOOLS)
      .then((name) => console.log(`Added Extension: ${name}`))
      .catch((err) => console.log('An error occurred: ', err))
  }
})

app.on('window-all-closed', () => {
  if (process.platform !== 'darwin') {
    app.quit()
  }
})
```

### **2. 설정 관리 시스템**

#### **설정 관리자** (`src/main/models/config/index.ts`)

```typescript
import fs from 'fs-extra'
import { getConfigPath, getWorkspacePath } from '../../common/paths/index.js'

export interface AppConfig {
  version: string
  workspacePath: string
  createdAt: string
  lastUsed: string
  // 향후 확장: 테마, 언어, 윈도우 상태 등
}

export interface WorkspaceSettings {
  version: string
  createdAt: string
  // 향후 확장: 프로젝트별 설정
}

/**
 * 앱 설정 로드
 */
export async function loadAppConfig(): Promise<AppConfig | null> {
  try {
    const configPath = getConfigPath()
    if (!(await fs.pathExists(configPath))) {
      return null
    }

    const config = await fs.readJson(configPath)

    // 설정 검증
    if (!config.workspacePath || !config.version) {
      throw new Error('Invalid configuration file')
    }

    // 마지막 사용 시간 업데이트
    config.lastUsed = new Date().toISOString()
    await saveAppConfig(config)

    return config
  } catch (error) {
    console.error('Failed to load app config:', error)
    return null
  }
}

/**
 * 앱 설정 저장
 */
export async function saveAppConfig(config: Partial<AppConfig>): Promise<void> {
  try {
    const configPath = getConfigPath()

    // 기존 설정과 병합
    let existingConfig: Partial<AppConfig> = {}
    if (await fs.pathExists(configPath)) {
      existingConfig = await fs.readJson(configPath)
    }

    const newConfig: AppConfig = {
      version: '1.0.0',
      createdAt: new Date().toISOString(),
      lastUsed: new Date().toISOString(),
      ...existingConfig,
      ...config,
    }

    await fs.writeJson(configPath, newConfig, { spaces: 2 })
  } catch (error) {
    console.error('Failed to save app config:', error)
    throw new Error(`Configuration save failed: ${error.message}`)
  }
}

/**
 * 워크스페이스 설정 로드
 */
export async function loadWorkspaceSettings(): Promise<WorkspaceSettings | null> {
  try {
    const workspacePath = getWorkspacePath()
    const settingsPath = join(workspacePath, 'settings.json')

    if (!(await fs.pathExists(settingsPath))) {
      return null
    }

    return await fs.readJson(settingsPath)
  } catch (error) {
    console.error('Failed to load workspace settings:', error)
    return null
  }
}

/**
 * 설정 무결성 검증
 */
export async function validateConfiguration(): Promise<{
  isValid: boolean
  errors: string[]
  warnings: string[]
}> {
  const result = {
    isValid: true,
    errors: [] as string[],
    warnings: [] as string[],
  }

  try {
    // 1. 앱 설정 검증
    const config = await loadAppConfig()
    if (!config) {
      result.errors.push('App configuration not found')
      result.isValid = false
      return result
    }

    // 2. 워크스페이스 경로 접근 확인
    const workspacePath = config.workspacePath
    if (!(await fs.pathExists(workspacePath))) {
      result.errors.push(`Workspace directory not found: ${workspacePath}`)
      result.isValid = false
    } else {
      try {
        await fs.access(workspacePath, fs.constants.R_OK | fs.constants.W_OK)
      } catch {
        result.errors.push(`No read/write permission for workspace: ${workspacePath}`)
        result.isValid = false
      }
    }

    // 3. 워크스페이스 설정 검증
    const workspaceSettings = await loadWorkspaceSettings()
    if (!workspaceSettings) {
      result.warnings.push('Workspace settings file missing (will be recreated)')
    }

    // 4. 필요한 디렉터리 존재 확인
    const requiredDirs = ['nodes', 'content', 'log']
    for (const dir of requiredDirs) {
      const dirPath = join(workspacePath, dir)
      if (!(await fs.pathExists(dirPath))) {
        result.warnings.push(`Directory missing: ${dir} (will be recreated)`)
      }
    }
  } catch (error) {
    result.errors.push(`Configuration validation failed: ${error.message}`)
    result.isValid = false
  }

  return result
}
```

### **3. 에러 복구 및 마이그레이션**

#### **복구 관리자** (`src/main/models/recovery/index.ts`)

```typescript
import fs from 'fs-extra'
import { join } from 'path'
import { app } from 'electron'
import { getConfigPath, getCachePath } from '../../common/paths/index.js'

/**
 * 손상된 설정 복구
 */
export async function recoverCorruptedConfig(): Promise<boolean> {
  try {
    const configPath = getConfigPath()
    const backupPath = `${configPath}.backup`

    // 백업 파일 확인
    if (await fs.pathExists(backupPath)) {
      console.log('Attempting to restore from backup...')
      await fs.copy(backupPath, configPath)
      return true
    }

    // 백업이 없으면 삭제하고 새로 시작
    if (await fs.pathExists(configPath)) {
      await fs.remove(configPath)
    }

    return false
  } catch (error) {
    console.error('Failed to recover corrupted config:', error)
    return false
  }
}

/**
 * 설정 백업 생성
 */
export async function createConfigBackup(): Promise<void> {
  try {
    const configPath = getConfigPath()
    if (await fs.pathExists(configPath)) {
      const backupPath = `${configPath}.backup`
      await fs.copy(configPath, backupPath)
    }
  } catch (error) {
    console.error('Failed to create config backup:', error)
  }
}

/**
 * 앱 상태 완전 초기화 (개발용/문제 해결용)
 */
export async function resetAppState(): Promise<void> {
  try {
    const userDataPath = app.getPath('userData')

    // 설정 파일들 삭제
    const configPath = getConfigPath()
    const cachePath = getCachePath()

    if (await fs.pathExists(configPath)) {
      await fs.remove(configPath)
    }

    if (await fs.pathExists(cachePath)) {
      await fs.remove(cachePath)
    }

    // 백업 파일들도 삭제
    const backupPath = `${configPath}.backup`
    if (await fs.pathExists(backupPath)) {
      await fs.remove(backupPath)
    }

    console.log('App state reset completed')
  } catch (error) {
    console.error('Failed to reset app state:', error)
    throw error
  }
}

/**
 * 설정 마이그레이션 (버전 업그레이드 시)
 */
export async function migrateConfiguration(fromVersion: string, toVersion: string): Promise<void> {
  console.log(`Migrating configuration from ${fromVersion} to ${toVersion}`)

  try {
    const config = await loadAppConfig()
    if (!config) return

    // 버전별 마이그레이션 로직
    if (fromVersion === '0.9.x' && toVersion === '1.0.0') {
      // 예시: 경로 구조 변경, 새 설정 추가 등
      config.version = toVersion
      await saveAppConfig(config)
    }

    console.log('Configuration migration completed')
  } catch (error) {
    console.error('Configuration migration failed:', error)
    throw error
  }
}
```

```typescript
export enum AppState {
  INITIALIZING = 'initializing',
  READY = 'ready',
  FIRST_RUN = 'first_run',
  ERROR = 'error',
  SHUTTING_DOWN = 'shutting_down',
}

export interface AppStateManager {
  currentState: AppState
  listeners: Set<(state: AppState) => void>
}

const appStateManager: AppStateManager = {
  currentState: AppState.INITIALIZING,
  listeners: new Set(),
}

/**
 * 앱 상태 변경
 */
export function setAppState(newState: AppState): void {
  const oldState = appStateManager.currentState
  appStateManager.currentState = newState

  console.log(`App state changed: ${oldState} → ${newState}`)

  // 리스너들에게 알림
  appStateManager.listeners.forEach((listener) => {
    try {
      listener(newState)
    } catch (error) {
      console.error('App state listener error:', error)
    }
  })
}

/**
 * 상태 변경 리스너 등록
 */
export function onAppStateChange(listener: (state: AppState) => void): () => void {
  appStateManager.listeners.add(listener)

  // 구독 해제 함수 반환
  return () => {
    appStateManager.listeners.delete(listener)
  }
}

/**
 * 현재 앱 상태 조회
 */
export function getAppState(): AppState {
  return appStateManager.currentState
}
```

```typescript
export interface AppMetrics {
  startupTime: number
  initializationSteps: Array<{
    step: string
    duration: number
    timestamp: number
  }>
  errors: Array<{
    error: string
    timestamp: number
    context: string
  }>
}

const metrics: AppMetrics = {
  startupTime: 0,
  initializationSteps: [],
  errors: [],
}

/**
 * 단계별 성능 측정
 */
export function measureStep<T>(stepName: string, fn: () => Promise<T>): Promise<T> {
  const startTime = performance.now()

  return fn().finally(() => {
    const duration = performance.now() - startTime
    metrics.initializationSteps.push({
      step: stepName,
      duration,
      timestamp: Date.now(),
    })

    console.log(`Step "${stepName}" completed in ${duration.toFixed(2)}ms`)
  })
}

/**
 * 메트릭스 리포트 생성
 */
export function generateMetricsReport(): AppMetrics {
  return { ...metrics }
}
```

```javascript
const { spawn } = require('child_process')
const fs = require('fs-extra')

/**
 * 빌드 결과 검증
 */
async function validateBuild() {
  console.log('Validating build...')

  // 필수 파일 존재 확인
  const requiredFiles = ['dist/main/index.js', 'dist/preload/index.js', 'dist/renderer/index.html']

  for (const file of requiredFiles) {
    if (!(await fs.pathExists(file))) {
      throw new Error(`Required file missing: ${file}`)
    }
  }

  // 앱 시작 테스트
  await testAppLaunch()

  console.log('Build validation completed successfully')
}

async function testAppLaunch() {
  return new Promise((resolve, reject) => {
    const app = spawn('electron', ['dist/main/index.js'], { timeout: 10000 })

    let hasLaunched = false

    app.on('spawn', () => {
      hasLaunched = true
      setTimeout(() => {
        app.kill()
        resolve()
      }, 3000)
    })

    app.on('error', reject)
    app.on('exit', (code) => {
      if (!hasLaunched && code !== 0) {
        reject(new Error(`App failed to launch with code ${code}`))
      }
    })
  })
}

validateBuild().catch(console.error)
```
