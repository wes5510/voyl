**역할**: 애플리케이션 전체 초기화 프로세스를 조정, 각 모듈의 초기화를 순서대로 호출

```typescript
 * 애플리케이션 전체 초기화
 */
async function initializeApplication(): Promise<void> {
  try {
    // 1. 앱 설정 확인
    if (!appModel.hasConfig()) {
      // UI를 통한 사용자 워크스페이스 경로 요청
      await requestUserWorkspacePath()
      return
    }

    // 2. 앱 초기화 (설정 로드 + 워크스페이스 초기화)
    const workspacePath = await appModel.initialize()

    // 3. DB 초기화 (경로를 인자로 전달)

    // 4. IPC 핸들러 등록 (초기화 완료 후)
    await import('./ipc/index.js')
  } catch (error) {
    console.error('Failed to initialize application:', error)
    app.quit()
  }
}

  // 앱 초기화 먼저 수행
  await initializeApplication()

  // 초기화 완료 후 창 생성
  createWindow()

  // ... 나머지 로직
})
```

```typescript
}
export async function getLogPath(): Promise<string> {
  /* ... */

/**
 * 워크스페이스 전체 초기화 조정자
 * 실패 시 자동 정리 수행
 */
export async function initializeWorkspace(workspacePath: string): Promise<void> {
  try {
    // 순차적 초기화 (의존성 고려)
    await validateWorkspacePath(workspacePath)
    await initDirectories(workspacePath)
    await initSettings(workspacePath)
  } catch (error) {
    // 초기화 실패 시 정리
    await cleanupFailedInitialization(workspacePath)
    throw error
  }
}

/**
 * 워크스페이스 경로 검증
 */
async function validateWorkspacePath(workspacePath: string): Promise<void> {
  // 경로 존재 확인
  if (!(await fs.pathExists(workspacePath))) {
    await fs.ensureDir(workspacePath)
  }

  // 읽기/쓰기 권한 확인
  try {
    await fs.access(workspacePath, fs.constants.R_OK | fs.constants.W_OK)
  } catch {
    throw new Error(`No read/write permission for: ${workspacePath}`)
  }
}

/**
 * 실패한 초기화 정리
 */
async function cleanupFailedInitialization(workspacePath: string): Promise<void> {
  try {
    const settingsPath = join(workspacePath, WORKSPACE_PATHS.SETTINGS_FILE)
    if (await fs.pathExists(settingsPath)) {
      await fs.remove(settingsPath)
    }
    // 빈 폴더들 정리 (노드/콘텐츠/로그)
    for (const dir of [
      WORKSPACE_PATHS.NODES_DIR,
      WORKSPACE_PATHS.CONTENT_DIR,
      WORKSPACE_PATHS.LOG_DIR,
    ]) {
      const dirPath = join(workspacePath, dir)
      if (await fs.pathExists(dirPath)) {
        const files = await fs.readdir(dirPath)
        if (files.length === 0) {
          await fs.remove(dirPath)
        }
      }
    }
  } catch {
    // 정리 실패는 무시 (로그만 남김)
    console.warn('Failed to cleanup incomplete workspace initialization')
  }
}
```

### **4. DB 초기화 구현** (`src/main/db/index.ts`)

**역할**: 캐시 DB의 완전한 재초기화 및 워크스페이스 데이터 로드

```typescript
import Database from 'better-sqlite3'
import { drizzle } from 'drizzle-orm/better-sqlite3'
import fs from 'fs-extra'
import { join } from 'path'
// ✅ 외부 의존성 없음, 모든 경로를 인자로 받음

// 현재 DB 인스턴스 (재초기화 가능하도록 변수로 관리)
let sqlite: Database | null = null
let db: any = null

/**
 * DB 완전 초기화 (다중 기기 동기화 대응)
 * @param options 초기화 옵션
 * @param options.workspacePath 워크스페이스 경로
 * @param options.cachePath 캐시 DB 파일 경로
 */
export async function initialize({
  workspacePath,
  cachePath,
}: {
  workspacePath: string
  cachePath: string
}): Promise<void> {
  try {
    // 1. 기존 DB 연결 정리
    await closeDatabase()

    // 2. 캐시 파일 삭제 (다중 기기 동기화 대응)
    if (await fs.pathExists(cachePath)) {
      await fs.remove(cachePath)
    }

    // 3. 새로운 DB 인스턴스 생성
    await createDatabase(cachePath)

    // 4. 스키마 생성/마이그레이션
    await createSchema()

    // 5. 워크스페이스 데이터 로드
    await loadWorkspaceData(workspacePath)
  } catch (error) {
    console.error('Failed to initialize database:', error)
    throw error
  }
}

/**
 * 기존 DB 연결 정리
 */
async function closeDatabase(): Promise<void> {
  if (sqlite) {
    sqlite.close()
    sqlite = null
    db = null
  }
}

/**
 * 새로운 DB 인스턴스 생성
 */
async function createDatabase(cachePath: string): Promise<void> {
  sqlite = new Database(cachePath)
  sqlite.pragma('journal_mode = WAL')

  db = drizzle(sqlite, {
    logger: process.env.NODE_ENV === 'development',
  })
}

/**
 * 스키마 생성/마이그레이션
 *
 * 🎯 권장 방법: drizzle-kit push 활용 (개발 시)
 * 🚀 런타임 방법: 기존 스키마 import + CREATE TABLE (현재 구현)
 */
async function createSchema(): Promise<void> {
  try {
    // 🎯 최적 방법: drizzle-kit push 사용
    // Production 환경에서는 사전에 `npx drizzle-kit push`로 스키마 생성
    // 하지만 cache DB는 매번 재생성되므로 런타임 생성 필요

    await createSchemaRuntime()

    console.log('✅ Database schema created successfully')
  } catch (error) {
    console.error('Failed to create schema:', error)
    throw error
  }
}

/**
 * 런타임 스키마 생성
 *
 * ✅ 확정된 최고의 방법: Drizzle migrate 함수!
 */
async function createSchemaRuntime(): Promise<void> {
  try {
    // Drizzle 공식 migrate 함수 사용 (production-ready!)
    const { migrate } = await import('drizzle-orm/better-sqlite3/migrator')

    await migrate(db, {
      migrationsFolder: './src/main/db/migrations',
    })

    console.log('✅ Schema migrated successfully with Drizzle migrate')
  } catch (error) {
    console.error('Migration failed:', error)
    throw error
  }
}

/**
 * 워크스페이스 데이터를 캐시 DB에 로드
 */
async function loadWorkspaceData(workspacePath: string): Promise<void> {
  try {
    // 1. settings.json에서 nodeTypes와 attributes 로드
    await loadSettingsData(workspacePath)

    // 2. nodes/ 디렉터리에서 모든 노드 파일 로드
    await loadNodesData(workspacePath)

    console.log(`✅ Workspace data loaded from ${workspacePath}`)
  } catch (error) {
    console.error('Failed to load workspace data:', error)
    throw error
  }
}

/**
 * settings.json에서 설정 데이터 로드
 */
async function loadSettingsData(workspacePath: string): Promise<void> {
  const settingsPath = join(workspacePath, 'settings.json')

  if (!(await fs.pathExists(settingsPath))) {
    console.log('No settings.json found, skipping settings data load')
    return
  }

  const settings = await fs.readJson(settingsPath)

  // nodeTypes 데이터 삽입
  if (settings.nodeTypes && Array.isArray(settings.nodeTypes)) {
    const insertNodeType = sqlite.prepare(`
      INSERT OR REPLACE INTO node_types (id, name, defined_attribute_names)
      VALUES (?, ?, ?)
    `)

    for (const nodeType of settings.nodeTypes) {
      insertNodeType.run(
        nodeType.id,
        nodeType.name,
        JSON.stringify(nodeType.definedAttributeNames || []),
      )
    }
    console.log(`Loaded ${settings.nodeTypes.length} nodeTypes`)
  }

  // attributes 데이터 삽입
  if (settings.attributes && Array.isArray(settings.attributes)) {
    const insertAttribute = sqlite.prepare(`
      INSERT OR REPLACE INTO attributes (id, name, value)
      VALUES (?, ?, ?)
    `)

    for (const attribute of settings.attributes) {
      insertAttribute.run(
        attribute.id,
        attribute.name,
        attribute.value ? JSON.stringify(attribute.value) : null,
      )
    }
    console.log(`Loaded ${settings.attributes.length} attributes`)
  }
}

/**
 * nodes/ 디렉터리에서 노드 파일들 로드
 */
async function loadNodesData(workspacePath: string): Promise<void> {
  const nodesDir = join(workspacePath, 'nodes')

  if (!(await fs.pathExists(nodesDir))) {
    console.log('No nodes/ directory found, skipping nodes data load')
    return
  }

  const nodeFiles = await fs.readdir(nodesDir)
  const jsonFiles = nodeFiles.filter((file) => file.endsWith('.json'))

  if (jsonFiles.length === 0) {
    console.log('No node JSON files found')
    return
  }

  const insertNode = sqlite.prepare(`
    INSERT OR REPLACE INTO nodes (id, parent_id, index, attribute_ids, title, content, type_id)
    VALUES (?, ?, ?, ?, ?, ?, ?)
  `)

  let loadedCount = 0
  for (const fileName of jsonFiles) {
    try {
      const filePath = join(nodesDir, fileName)
      const nodeData = await fs.readJson(filePath)

      insertNode.run(
        nodeData.id,
        nodeData.parentId || null,
        nodeData.index || '',
        JSON.stringify(nodeData.attributeIds || []),
        nodeData.title || '',
        nodeData.content || '',
        nodeData.typeId || null,
      )
      loadedCount++
    } catch (fileError) {
      console.warn(`Failed to load node file ${fileName}:`, fileError)
    }
  }

  console.log(`Loaded ${loadedCount}/${jsonFiles.length} nodes`)
}

/**
 * DB 인스턴스 반환 (기존 connect.ts 대체)
 */
export function getDatabase() {
  if (!db) {
    throw new Error('Database not initialized. Call initialize() first.')
  }
  return db
}
```

- **models/app/workspace/index.ts**: 워크스페이스 비즈니스 로직(`initializeWorkspace`) + 워크스페이스 경로 함수들
- **db/index.ts**: DB 완전 초기화(`initialize`) + DB 연결 관리 + 스키마 생성 + 데이터 로드 (완전 독립적, 외부 의존성 없음)
- **완전한 캡슐화**: main은 각 모듈의 high-level API만 호출, 내부 구현 완전 숨김
- **ESLint 규칙 준수**: models는 db와 common만 import, 초기화 조정은 main에서
- **단순화된 구조**: 각 레벨에서 const.ts + index.ts만 사용

```typescript
import { tmpdir } from 'os'
import { join } from 'path'
import fs from 'fs-extra'
import { initializeWorkspace } from './index.js'

describe('Workspace Initialization', () => {
  const testWorkspacePath = join(tmpdir(), 'test-workspace') // ✅ 크로스 플랫폼 호환

  beforeEach(async () => {
    await fs.remove(testWorkspacePath)
  })

  describe('initializeWorkspace', () => {
    it('should create all required directories', async () => {
      await initializeWorkspace(testWorkspacePath)

      expect(await fs.pathExists(join(testWorkspacePath, 'nodes'))).toBe(true)
      expect(await fs.pathExists(join(testWorkspacePath, 'content'))).toBe(true)
      expect(await fs.pathExists(join(testWorkspacePath, 'log'))).toBe(true)
    })

    it('should create settings.json with default values', async () => {
      await initializeWorkspace(testWorkspacePath)

      const settingsPath = join(testWorkspacePath, 'settings.json')
      const settings = await fs.readJson(settingsPath)
      expect(settings.version).toBe('1.0.0')
      expect(settings.createdAt).toBeDefined()
    })

    it('should cleanup on failure', async () => {
      // Mock fs.ensureDir to fail
      jest.spyOn(fs, 'ensureDir').mockRejectedValueOnce(new Error('Permission denied'))

      await expect(initializeWorkspace(testWorkspacePath)).rejects.toThrow()

      // Verify cleanup (설정 파일 제거 확인)
      const settingsPath = join(testWorkspacePath, 'settings.json')
      expect(await fs.pathExists(settingsPath)).toBe(false)
    })
  })
})
```
