    // 1. Config 파일 존재 여부 확인

      // 사용자 액션 필요 (UI 트리거)

### **2. 워크스페이스 초기화 조정자** (`src/main/models/workspace/index.ts`)

**역할**: 워크스페이스 관련 모든 초기화 작업을 순서대로 조정

```typescript
import fs from 'fs-extra'
import { join } from 'path'
import { PATHS } from '../../common/paths/const.js'
import { getCachePath } from '../../common/paths/index.js'

/**
 * 워크스페이스 전체 초기화 조정자
 * 실패 시 자동 정리 수행
 */
export async function initializeWorkspace(workspacePath: string): Promise<void> {
  try {
    // 순차적 초기화 (의존성 고려)
    await validateWorkspacePath(workspacePath)
    await initDirectories(workspacePath)
    await initSettings(workspacePath)
    await initCacheDB(workspacePath)
  } catch (error) {
    // 초기화 실패 시 정리
    await cleanupFailedInitialization(workspacePath)
    throw error
  }
}

/**
 * 워크스페이스 경로 검증
 */
async function validateWorkspacePath(workspacePath: string): Promise<void> {
  // 경로 존재 확인
  if (!(await fs.pathExists(workspacePath))) {
    await fs.ensureDir(workspacePath)
  }

  // 읽기/쓰기 권한 확인
  try {
    await fs.access(workspacePath, fs.constants.R_OK | fs.constants.W_OK)
  } catch {
    throw new Error(`No read/write permission for: ${workspacePath}`)
  }
}

/**
 * 실패한 초기화 정리
 */
async function cleanupFailedInitialization(workspacePath: string): Promise<void> {
  try {
    const settingsPath = join(workspacePath, PATHS.WORKSPACE.SETTINGS_FILE)
    if (await fs.pathExists(settingsPath)) {
      await fs.remove(settingsPath)
    }
    // 빈 폴더들 정리 (노드/콘텐츠/로그)
    for (const dir of [
      PATHS.WORKSPACE.NODES_DIR,
      PATHS.WORKSPACE.CONTENT_DIR,
      PATHS.WORKSPACE.LOG_DIR,
    ]) {
      const dirPath = join(workspacePath, dir)
      if (await fs.pathExists(dirPath)) {
        const files = await fs.readdir(dirPath)
        if (files.length === 0) {
          await fs.remove(dirPath)
        }
      }
    }
  } catch {
    // 정리 실패는 무시 (로그만 남김)
    console.warn('Failed to cleanup incomplete workspace initialization')
  }
}
```

### **3. 개별 초기화 함수들**

#### **디렉터리 생성** (`initDirectories`)

```typescript
/**
 * 워크스페이스 내 필요한 디렉터리들을 병렬 생성
 */
export async function initDirectories(workspacePath: string): Promise<void> {
  const directories = [
    join(workspacePath, PATHS.WORKSPACE.NODES_DIR),
    join(workspacePath, PATHS.WORKSPACE.CONTENT_DIR),
    join(workspacePath, PATHS.WORKSPACE.LOG_DIR),
  ]

  // 병렬 생성으로 성능 최적화
  await Promise.all(directories.map((dir) => fs.ensureDir(dir)))
}
```

#### **설정 파일 초기화** (`initSettings`)

```typescript
/**
 * 워크스페이스 설정 파일 생성/검증
 */
export async function initSettings(workspacePath: string): Promise<void> {
  const settingsPath = join(workspacePath, PATHS.WORKSPACE.SETTINGS_FILE)

  if (await fs.pathExists(settingsPath)) {
    // 기존 설정 검증
    await validateExistingSettings(settingsPath)
  } else {
    // 새 설정 생성
    const defaultSettings = createDefaultSettings()
    await fs.writeJson(settingsPath, defaultSettings, { spaces: 2 })
  }
}

/**
 * 기본 워크스페이스 설정 생성
 */
function createDefaultSettings() {
  return {
    version: '1.0.0',
    createdAt: new Date().toISOString(),
    // 향후 확장: 테마, 언어, 플러그인 등
  }
}
```

#### **캐시 DB 초기화** (`initCacheDB`)

```typescript
/**
 * SQLite 캐시 데이터베이스 초기화
 * connect.ts와 연동하여 스키마 생성
 */
export async function initCacheDB(workspacePath: string): Promise<void> {
  // 캐시 DB는 앱 전역이므로 워크스페이스 독립적
  const cachePath = getCachePath()

  if (!(await fs.pathExists(cachePath))) {
    // DB 파일이 없으면 connect.ts가 자동 생성하므로
    // 여기서는 경로 접근 가능성만 확인
    const cacheDir = path.dirname(cachePath)
    await fs.ensureDir(cacheDir)
  }
}
```

describe('Workspace Initialization', () => {
  const testWorkspacePath = '/tmp/test-workspace'

  beforeEach(async () => {
    await fs.remove(testWorkspacePath)
  })

  describe('initializeWorkspace', () => {
    it('should create all required directories', async () => {
      await initializeWorkspace(testWorkspacePath)

      expect(await fs.pathExists(join(testWorkspacePath, 'nodes'))).toBe(true)

      const settingsPath = join(testWorkspacePath, 'settings.json')
      const settings = await fs.readJson(settingsPath)
      expect(settings.version).toBe('1.0.0')
      expect(settings.createdAt).toBeDefined()

      jest.spyOn(fs, 'ensureDir').mockRejectedValueOnce(new Error('Permission denied'))

      await expect(initializeWorkspace(testWorkspacePath)).rejects.toThrow()

      // Verify cleanup
      expect(await fs.pathExists(testWorkspacePath)).toBe(false)
    })
  })

## ⚙️ 에러 처리 전략

### **에러 타입별 구분**

| 에러 타입            | 처리 방법           | 사용자 경험               |
| -------------------- | ------------------- | ------------------------- |
| **CONFIG_NOT_FOUND** | UI 트리거           | 저장 위치 선택 모달       |
| **권한 없음**        | 앱 중단             | 에러 다이얼로그 + 종료    |
| **디스크 공간 부족** | 앱 중단             | 에러 다이얼로그 + 종료    |
| **부분 초기화 실패** | 자동 정리 후 재시도 | 진행률 표시 + 재시도 버튼 |

### **복구 메커니즘**

1. **자동 정리**: 실패 시 생성된 파일/폴더 제거
2. **상태 검증**: 기존 설정 파일 무결성 확인
3. **점진적 복구**: 부분적 성공 상태에서 재시도 지원
