### **2. 추천 경로 제공 핸들러**

**목표**: OS별 사용자 친화적 기본 경로 제안

```typescript
// src/main/ipc/storage.ts
import { IpcMain, dialog } from 'electron'
import { homedir } from 'os'
import { join } from 'path'
import fs from 'fs-extra'

interface RecommendedPath {
  name: string
  path: string
  description: string
  isCloudService: boolean
  icon?: string
}

/**
 * OS별 추천 저장 경로 제공
 */
export async function getRecommendedPaths(): Promise<RecommendedPath[]> {
  const home = homedir()
  const platform = process.platform

  const paths: RecommendedPath[] = [
    // 기본 문서 폴더
    {
      name: 'Documents',
      path: join(home, 'Documents', 'Voyl'),
      description: '문서 폴더에 Voyl 워크스페이스 생성',
      isCloudService: false,
      icon: 'folder',
    },
  ]

  // 클라우드 서비스 자동 감지
  const cloudPaths = await detectCloudServices(home, platform)
  paths.push(...cloudPaths)

  // 커스텀 경로 선택 옵션
  paths.push({
    name: 'Custom Location',
    path: '',
    description: '다른 위치 선택...',
    isCloudService: false,
    icon: 'folder-open',
  })

  return paths
}

/**
 * 클라우드 서비스 폴더 자동 감지
 */
async function detectCloudServices(home: string, platform: string): Promise<RecommendedPath[]> {
  const cloudServices = [
    {
      name: 'iCloud Drive',
      paths: [join(home, 'Library', 'Mobile Documents', 'com~apple~CloudDocs')],
      description: 'iCloud로 동기화',
      platforms: ['darwin'],
    },
    {
      name: 'Dropbox',
      paths: [join(home, 'Dropbox'), join(home, 'Library', 'CloudStorage', 'Dropbox')],
      description: 'Dropbox로 동기화',
      platforms: ['darwin', 'win32', 'linux'],
    },
    {
      name: 'Google Drive',
      paths: [join(home, 'Library', 'CloudStorage', 'GoogleDrive-*'), join(home, 'Google Drive')],
      description: 'Google Drive로 동기화',
      platforms: ['darwin', 'win32'],
    },
    {
      name: 'OneDrive',
      paths: [join(home, 'Library', 'CloudStorage', 'OneDrive-*'), join(home, 'OneDrive')],
      description: 'OneDrive로 동기화',
      platforms: ['darwin', 'win32'],
    },
  ]

  const detected: RecommendedPath[] = []

  for (const service of cloudServices) {
    if (!service.platforms.includes(platform)) continue

    for (const pathPattern of service.paths) {
      try {
        // 와일드카드 처리
        if (pathPattern.includes('*')) {
          const basePath = pathPattern.split('*')[0]
          if (await fs.pathExists(basePath)) {
            const dirs = await fs.readdir(basePath)
            for (const dir of dirs) {
              const fullPath = join(basePath, dir)
              if (await fs.pathExists(fullPath)) {
                detected.push({
                  name: service.name,
                  path: join(fullPath, 'Voyl'),
                  description: service.description,
                  isCloudService: true,
                  icon: 'cloud',
                })
                break
              }
            }
          }
        } else {
          if (await fs.pathExists(pathPattern)) {
            detected.push({
              name: service.name,
              path: join(pathPattern, 'Voyl'),
              description: service.description,
              isCloudService: true,
              icon: 'cloud',
            })
            break
          }
        }
      } catch {
        // 접근 불가한 경로는 무시
        continue
      }
    }
  }

  return detected
}
```

### **3. 경로 검증 핸들러**

```typescript
interface PathValidationResult {
  isValid: boolean
  error?: string
  warnings: string[]
  suggestions?: string[]
}

/**
 * 사용자 선택 경로 검증
 */
export async function validateStoragePath(path: string): Promise<PathValidationResult> {
  const result: PathValidationResult = {
    isValid: false,
    warnings: [],
  }

  try {
    // 1. 경로 존재 확인
    if (!(await fs.pathExists(path))) {
      try {
        await fs.ensureDir(path)
      } catch (error) {
        result.error = `Cannot create directory: ${error.message}`
        return result
      }
    }

    // 2. 권한 확인
    try {
      await fs.access(path, fs.constants.R_OK | fs.constants.W_OK)
    } catch {
      result.error = 'No read/write permission for this location'
      return result
    }

    // 3. 시스템 폴더 방지
    const systemPaths = await getSystemPaths()
    if (systemPaths.some((sysPath) => path.startsWith(sysPath))) {
      result.error = 'Cannot use system directories'
      return result
    }

    // 4. 기존 Voyl 워크스페이스 확인
    const settingsPath = join(path, 'settings.json')
    if (await fs.pathExists(settingsPath)) {
      result.warnings.push('This directory already contains a Voyl workspace')
    }

    // 5. 디스크 공간 확인 (최소 100MB)
    const stats = await fs.stat(path)
    // Note: 정확한 디스크 공간 확인은 별도 라이브러리 필요

    result.isValid = true
    return result
  } catch (error) {
    result.error = `Path validation failed: ${error.message}`
    return result
  }
}

/**
 * OS별 시스템 경로 목록
 */
async function getSystemPaths(): Promise<string[]> {
  const platform = process.platform
  const home = homedir()

  switch (platform) {
    case 'darwin':
      return [
        '/System',
        '/Library',
        '/Applications',
        '/usr',
        '/bin',
        '/sbin',
        join(home, 'Library'),
        join(home, 'Applications'),
      ]
    case 'win32':
      return ['C:\\Windows', 'C:\\Program Files', 'C:\\Program Files (x86)', 'C:\\ProgramData']
    case 'linux':
      return ['/bin', '/sbin', '/usr', '/lib', '/etc', '/var', '/sys', '/proc']
    default:
      return []
  }
}
```

### **4. 워크스페이스 초기화 핸들러**

```typescript
import { initializeWorkspace } from '../models/workspace/index.js'
import { hasConfig, getConfigPath } from '../common/paths/index.js'

interface InitializationProgress {
  step: string
  progress: number // 0-100
  message: string
}

/**
 * 워크스페이스 초기화 (진행률 포함)
 */
export async function initializeWorkspaceWithProgress(
  workspacePath: string,
  onProgress?: (progress: InitializationProgress) => void,
): Promise<void> {
  const reportProgress = (step: string, progress: number, message: string) => {
    onProgress?.({ step, progress, message })
  }

  try {
    reportProgress('validation', 10, 'Validating workspace path...')

    // 1. 경로 검증
    const validation = await validateStoragePath(workspacePath)
    if (!validation.isValid) {
      throw new Error(validation.error)
    }

    reportProgress('config', 30, 'Creating app configuration...')

    // 2. 앱 설정 저장
    await saveAppConfig({ workspacePath })

    reportProgress('workspace', 60, 'Initializing workspace...')

    // 3. 워크스페이스 초기화
    await initializeWorkspace(workspacePath)

    reportProgress('complete', 100, 'Workspace initialized successfully')
  } catch (error) {
    // 실패 시 설정 정리
    try {
      if (await hasConfig()) {
        await fs.remove(getConfigPath())
      }
    } catch {
      // 정리 실패는 무시
    }
    throw error
  }
}

/**
 * 앱 설정 저장
 */
async function saveAppConfig(config: { workspacePath: string }): Promise<void> {
  const configPath = getConfigPath()
  await fs.writeJson(configPath, config, { spaces: 2 })
}
```

```typescript
/**
 * 디렉터리 선택 다이얼로그
 */
export async function openDirectoryDialog(): Promise<string | null> {
  const result = await dialog.showOpenDialog({
    properties: ['openDirectory', 'createDirectory'],
    title: 'Select Workspace Location',
    buttonLabel: 'Select',
    defaultPath: homedir(),
  })

  if (result.canceled || result.filePaths.length === 0) {
    return null
  }

  return result.filePaths[0]
}
```

## 🏗️ 핸들러 등록

```typescript
// src/main/ipc/storage.ts
export default function storageHandlers(ipcMain: IpcMain) {
  // 추천 경로
  ipcMain.handle(STORAGE_CHANNELS.GET_RECOMMENDED_PATHS, getRecommendedPaths)

  // 경로 검증
  ipcMain.handle(STORAGE_CHANNELS.VALIDATE_PATH, (_, { path }) => validateStoragePath(path))

  // 워크스페이스 초기화
  ipcMain.handle(STORAGE_CHANNELS.INITIALIZE_WORKSPACE, (_, { path }, onProgress) =>
    initializeWorkspaceWithProgress(path, onProgress),
  )

  // 설정 존재 확인
  ipcMain.handle(STORAGE_CHANNELS.CHECK_CONFIG_EXISTS, () => hasConfig())

  // 디렉터리 다이얼로그
  ipcMain.handle(STORAGE_CHANNELS.OPEN_DIRECTORY_DIALOG, openDirectoryDialog)
}

// src/main/ipc/index.ts에 추가
import storageHandlers from './storage.js'

storageHandlers(ipcMain)
```
